=== makefile ===
.PHONY: dev worker test up down clean

PYTHONPATH=.

dev:
	uvicorn api.main:app --reload --host 0.0.0.0 --port 8000

worker:
	celery -A workers.celery_app worker -l info

test:
	PYTHONPATH=. pytest

up:
	@echo "Note: Docker is deferred. Use 'make dev' and 'make worker' for local development."

down:
	@echo "Note: Docker is deferred."

clean:
	find . -type d -name "__pycache__" -exec rm -rf {} +
	find . -type f -name "*.pyc" -delete
	rm -f hajeen.sqlite3

ENV PYTHONUNBUFFERED=1 PYTHONDONTWRITEBYTECODE=1

    build-essential libpq-dev curl && rm -rf /var/lib/apt/lists/*

RUN pip install --no-cache-dir --prefix=/install -r base.txt -r api.txt

FROM python:3.11-slim AS production

ENV PYTHONUNBUFFERED=1 \
    PYTHONDONTWRITEBYTECODE=1 \
    PORT=8000 \
    WORKERS=4

RUN apt-get update && apt-get install -y --no-install-recommends \
    libpq5 curl && rm -rf /var/lib/apt/lists/*

RUN groupadd -r apiuser && useradd -r -g apiuser apiuser

COPY --from=builder /install /usr/local
WORKDIR /app
COPY --chown=apiuser:apiuser . .

USER apiuser
EXPOSE 8000

HEALTHCHECK --interval=15s --timeout=5s --start-period=10s --retries=3 \
    CMD curl -f http://localhost:8000/api/v1/health || exit 1

CMD ["sh", "-c", "gunicorn api.main:app --workers $WORKERS --worker-class uvicorn.workers.UvicornWorker --bind 0.0.0.0:$PORT --access-logfile - --error-logfile -"]
